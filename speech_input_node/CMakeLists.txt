cmake_minimum_required(VERSION 3.16)
project(speech_input_node LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(xiaocai_msgs REQUIRED)
find_package(Threads REQUIRED)
set(VENDOR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../thirdparty")
include_directories(include "${VENDOR_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../common/include")
add_executable(speech_input_node src/speech_input.cpp)
ament_target_dependencies(speech_input_node rclcpp std_msgs xiaocai_msgs)
target_link_libraries(speech_input_node Threads::Threads)
target_link_libraries(speech_input_node "${VENDOR_DIR}/libvosk.so")
target_link_libraries(speech_input_node "${VENDOR_DIR}/libsherpa-onnx-c-api.so")
install(TARGETS speech_input_node DESTINATION lib/${PROJECT_NAME})
ament_package()
