cmake_minimum_required(VERSION 3.16)
project(music_node LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(xiaocai_msgs REQUIRED)
find_package(Threads REQUIRED)
include_directories(include "${CMAKE_CURRENT_SOURCE_DIR}/../thirdparty" "${CMAKE_CURRENT_SOURCE_DIR}/../common/include")
add_executable(music_node src/music.cpp)
ament_target_dependencies(music_node rclcpp std_msgs xiaocai_msgs)
target_link_libraries(music_node Threads::Threads)
install(TARGETS music_node DESTINATION lib/${PROJECT_NAME})
ament_package()
